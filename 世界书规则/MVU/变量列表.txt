<%
// 只裁剪发给模型的副本；当前楼层的事件与结算历史原样保存。
const rkCurrent = getvar('stat_data');
const rkPromptState = rkCurrent && typeof rkCurrent === 'object' ? JSON.parse(JSON.stringify(rkCurrent)) : null;
const rkObject = value => value !== null && typeof value === 'object' && !Array.isArray(value);
if (rkPromptState) {
  delete rkPromptState.$internal;
  // 收据、事件及赛期外赛事数据只从提示副本省略。
  if (rkObject(rkPromptState.系统)) delete rkPromptState.系统.关系计分;
  if (rkObject(rkPromptState.场景)) {
    // 只保留最近 4 条已发生事件，更早的留在存档；不新增字段，避免模型误写。
    const events = rkPromptState.场景.已发生事件;
    if (rkObject(events)) {
      const keys = Object.keys(events), keep = keys.slice(-4), shown = {};
      for (const key of keep) shown[key] = events[key];
      if (keep.length) rkPromptState.场景.已发生事件 = shown;
      else delete rkPromptState.场景.已发生事件;
    }
  }
  if (rkObject(rkPromptState.玩家) && rkObject(rkPromptState.玩家.成长)) {
    const growth = rkPromptState.玩家.成长;
    rkPromptState.玩家.成长 = Object.hasOwn(growth, '经验') ? { 经验: growth.经验 } : {};
  }
}
%>
